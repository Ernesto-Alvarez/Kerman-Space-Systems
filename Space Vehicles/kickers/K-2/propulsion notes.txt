dV = Isp * 9.8 * ln ( m0 / mf )			Tsiolkovsky's rocket equation

Isp = 320
m0 = 2066
mf = 586

dV = 320 * 9.8 * ln ( (m0 + P) / (mf + P) )	Rocket equation for a kicker transporting a payload of mass P

dV / 320 * 9.8 = ln ( (m0 + P) / (mf + P) )

dV / 3136 = ln ( (m0 + P) / (mf + P) )

m0 = 2066
mf = 586

dV / 3136 = ln ( (2066 + P) / (586 + P) )

e^(dV / 3136) = (2066 + P) / (586 + P)

e^(dV / 3136) * (586 + P) = 2066 + P

e^(dV / 3136) * 586 + e^(dV / 3136) * P = 2066 + P

e^(dV / 3136) * 586 + e^(dV / 3136) * P - 2066 = P

e^(dV / 3136) * 586 -2066 = P - e^(dV / 3136) * P

e^(dV / 3136) * 586 -2066 = P (1 - e^(dV / 3136) )

[ e^(dV / 3136) * 586 - 2066 ] / [ (1 - e^(dV / 3136) ) ] = P 

dV = 2696
e^(dV / 3136) = 2.3624

[ 2.3624 * 586 - 2066 ] / [ 1 - 2.3624] = P = 500.3182


[ e^(dV / 3136) * mf - m0 ] / [ (1 - e^(dV / 3136) ) ] = P 	Inverse rocket equation, how much payload we can transport to a specific dV
